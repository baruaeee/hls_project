function y = neuron(x, w, b)
    % 1. Force complete block RAM partitioning so every element gets its own wire
    coder.hdl.literaltext("#pragma HLS ARRAY_PARTITION variable=x complete dim=1");
    coder.hdl.literaltext("#pragma HLS ARRAY_PARTITION variable=w complete dim=1");
    %#pragma HLS BIND_OP: This is a Vitis HLS instruction that tells the synthesizer
    % how to implement a specific arithmetic or logic operation in hardware.
    % variable=y: Specifies that this binding rule applies to the variable
    % y (the accumulator/output where the addition results are stored).
    % op=add: Specifies that the target operation is addition (you can also bind
    % multiplications using op=mul, etc.).
    % impl=fabric: This is the most crucial part. It tells Vitis HLS to implement
    % the addition using the FPGA's general-purpose logic fabric (LUTs and registers)
    % rather than dedicated hardware blocks like DSP slices (which are configured
    % as impl=dsp).
    % only addition is used in fabric LUT beacause with mutliply logic it takes
    % too manz HW resources and clock/speed performance will significantly be reduced
    coder.hdl.literaltext("#pragma HLS BIND_OP variable=y op=add impl=fabric");

    y = b;

    % 2. Fully unroll the loop 
    coder.hdl.loopspec('unroll');
